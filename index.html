/* ══════════════════════════════════════════════════════════ DATA FETCHING LOGIC ══════════════════════════════════════════════════════════ */ async function loadChartData() { if (!curTicker) return; showState('loading'); // Tính toán thời gian bắt đầu (fromTs) dựa trên số ngày cấu hình (rangeDays) const toTs = Math.floor(Date.now() / 1000); const fromTs = toTs - (rangeDays * 24 * 60 * 60); // Khởi tạo URL nguồn dữ liệu (Bạn có thể chọn 1 trong 3 nguồn dưới đây bằng cách bỏ comment) // Nguồn 1: Dữ liệu từ file local pre-generated (.json) const url = `data/${curTicker}.json`; // Nguồn 2: Nếu bạn muốn gọi API dạng TradingView (SSI / Entrade) // const url = `https://api.entrade.com.vn/chart/history?resolution=${resolution === 'D' ? '1D' : resolution}&symbol=${curTicker}&from=${fromTs}&to=${toTs}`; // Nguồn 3: Nếu bạn muốn gọi API từ TCBS // const url = `https://tcanalysis.tcbs.com.vn/api/v1/bars?ticker=${curTicker}&resolution=${resolution}&from=${fromTs}&to=${toTs}`; try { const response = await fetch(url); if (!response.ok) { throw new Error(`Không thể tải dữ liệu (Mã lỗi: ${response.status})`); } const json = await response.json(); let candles = null; // Tự động nhận diện cấu trúc dữ liệu trả về để dùng đúng Parser tương ứng if (json && Array.isArray(json.candles)) { candles = parseLocalFile(json, fromTs, resolution); } else if (json && Array.isArray(json.t)) { candles = parseTVFormat(json); } else if (json && Array.isArray(json.data)) { candles = parseTCBS(json); } // Kiểm tra kết quả sau khi parse if (!candles || candles.length === 0) { showState('error', 'Dữ liệu trả về trống hoặc không đúng định dạng.'); return; } // Đẩy dữ liệu vào Lightweight Charts candleSeries.setData(candles); // Cập nhật các đường chỉ báo MA20, MA50 dựa trên dữ liệu nến mới if (typeof renderMA === 'function') { renderMA(candles); } // Ẩn các trạng thái chờ, hiển thị biểu đồ chính thức showState('ready'); mainChart.timeScale().fitContent(); } catch (error) { console.error("Lỗi fetch dữ liệu:", error); showState('error', `Lỗi kết nối: ${error.message}`); } }
